\scriptsize

  \item \textbf{Adiabatic Slope \& State Forms:} $T_1 V_1^{\gamma-1} = T_2 V_2^{\gamma-1}$, $T_2/T_1 = (P_2/P_1)^{(\gamma-1)/\gamma}$. On $P$--$T$, adiabat slope gives $\dfrac{C_{V,m}}{R} = \dfrac{\ln(P_2/P_1)}{\ln(T_2/T_1)} - 1$ (\texttt{MIT 2019 P2b}).
  \item \textbf{Downward Chord $P(V) = P_{\mathrm{int}}(1 - V/V_{\mathrm{int}})$:} Peak temperature $T_{\max} = \dfrac{P_{\mathrm{int}}V_{\mathrm{int}}}{4nR}$ occurs at midpoint $(V_{\mathrm{int}}/2, P_{\mathrm{int}}/2)$; heat transfer switches sign ($\delta Q = 0$) at $V_{\mathrm{turn}} = \dfrac{\gamma}{\gamma+1}V_{\mathrm{int}}$.
  \item \textbf{Joule--Thomson Coefficient:} $\mu_{\mathrm{JT}} = \left(\dfrac{\Delta T}{\Delta P}\right)_H \approx \dfrac{1}{C_{P,m}}\left(\dfrac{2a}{RT} - b\right)$. Since throttling has $\Delta P < 0$: $\mu_{\mathrm{JT}} > 0 \implies$ \textbf{cools} ($\mathrm{N_2, O_2, CO_2}$ at $\SI{300}{\kelvin}$); $\mu_{\mathrm{JT}} < 0 \implies$ \textbf{heats} ($\mathrm{He, H_2}$ at $\SI{300}{\kelvin}$).
  \item \textbf{Steady-Flow Energy Equation (SFEE):} $\dot{Q}_{\mathrm{cv}} - \dot{W}_{s,\mathrm{by}} = \dot{m}\left[(h_2 - h_1) + \frac{1}{2}(v_2^2 - v_1^2) + g_n(z_2 - z_1)\right]$.
  \textbf{Nozzle:} $v_2 = \sqrt{v_1^2 + 2(h_1 - h_2)}$ (\emph{convert $\Delta h$ to $\si{\joule\per\kilogram}$!}). \textbf{Turbine:} $\dot{W}_t = \dot{m}(h_1 - h_2)$. \textbf{Evacuated Insulated Tank Fill:} $\boxed{T_{\mathrm{tank},2} = \gamma\,T_{\mathrm{line}}}$.

\begin{formulacard}[Steady-State Heat Conduction, Resistance Networks, Critical Radius \& Wiedemann--Franz \src{S12,S13,S25}]
\scriptsize
\begin{itemize}
  \item \textbf{Fourier's Law \& Plane Wall:} $\displaystyle \dot{Q}_{\mathrm{cond}} = k A\,\frac{T_H - T_C}{L} = \frac{T_H - T_C}{R_{\mathrm{slab}}}$, \quad $R_{\mathrm{slab}} = \dfrac{L}{kA}$ ($\si{\kelvin\per\watt}$), \quad Unit-area $R$-value: $R_{\mathrm{val}}'' = \dfrac{L}{k}$ ($\si{\square\meter\kelvin\per\watt}$).
  \item \textbf{Cylindrical Shell (length $l$, radii $r_i \to r_o$) \& Spherical Shell ($r_i \to r_o$) (Rule \texttt{5.d.iv} Algebraic Form):}
  \[
    \boxed{R_{\mathrm{cyl}} = \frac{\ln(r_o/r_i)}{2\pi k l}, \qquad \dot{Q}_{\mathrm{cyl}} = \frac{2\pi k l(T_i - T_o)}{\ln(r_o/r_i)}}, \qquad \qquad \boxed{R_{\mathrm{sph}} = \frac{r_o - r_i}{4\pi k r_i r_o} = \frac{1/r_i - 1/r_o}{4\pi k}, \qquad \dot{Q}_{\mathrm{sph}} = \frac{4\pi k r_i r_o(T_i - T_o)}{r_o - r_i}}
  \]
  \item \textbf{Series Composite Wall, Interface Temperature $T_{\mathrm{int}}$ \& Parallel Thermal Bridges:}
  \[
    R_{\mathrm{series}} = \frac{1}{h_{\mathrm{in}}A} + \sum_{i=1}^N \frac{L_i}{k_i A} + \frac{1}{h_{\mathrm{out}}A} + \frac{1}{h_c A}, \qquad \boxed{T_{\mathrm{int,1\text{--}2}} = \frac{R_2 T_H + R_1 T_C}{R_1 + R_2}}, \qquad \frac{1}{R_{\mathrm{parallel}}} = \sum_{j=1}^M \frac{1}{R_j} \implies k_{\mathrm{eff},\parallel} = \frac{\sum k_j A_j}{\sum A_j}
  \]
  \item \textbf{Critical Radius of Insulation ($r_{\mathrm{crit}}$, where total $R_{\mathrm{tot}}(r_o) = R_{\mathrm{cond}}(r_o) + \frac{1}{h A(r_o)}$ is MINIMIZED / heat loss MAXIMIZED):}
  \[
    \boxed{r_{\mathrm{crit,cylinder}} = \frac{k_{\mathrm{ins}}}{h}}, \qquad \boxed{r_{\mathrm{crit,sphere}} = \frac{2 k_{\mathrm{ins}}}{h}} \quad (\text{If } r_i < r_{\mathrm{crit}},\text{ adding insulation \textbf{increases} heat loss until } r_o > r_{\mathrm{crit}}\text{! For 2027 cup, } r_i=\SI{37.5}{\milli\meter} \gg r_{\mathrm{crit}}\approx\SI{6}{\milli\meter}.)
  \]
  \item \textbf{Kinetic Thermal Conductivity \& Wiedemann--Franz Law (\texttt{2027 Cornell E1 Q32--34}):}
  $\displaystyle k_{\mathrm{kin}} = \frac{1}{3}C_V v\lambda = \frac{3 n_e k_B^2 T\tau}{2 m_e}$, Drude electrical conductivity $\sigma_e = \dfrac{n_e e^2\tau}{m_e} \implies \boxed{\frac{k}{\sigma_e T} = L_{\mathrm{Lorenz}}}$, where $L_{\mathrm{classical}} = \dfrac{3}{2}\left(\dfrac{k_B}{e}\right)^2 = \SI{1.11e-8}{\watt\ohm\per\kelvin^2}$ and quantum Sommerfeld $L_{\mathrm{quantum}} = \dfrac{\pi^2}{3}\left(\dfrac{k_B}{e}\right)^2 = \mathbf{\SI{2.44e-8}{\watt\ohm\per\kelvin^2}}$.
\end{itemize}
\end{formulacard}

\begin{twocol}[0.49]
\begin{formulacard}[Convection, Dimensionless Groups \& Dead-Air Gaps \src{S12}]
\scriptsize
\begin{itemize}
  \item \textbf{Newton's Law of Cooling:} $\dot{Q}_{\mathrm{conv}} = h A(T_s - T_\infty) = \dfrac{T_s - T_\infty}{R_{\mathrm{conv}}}$, $R_{\mathrm{conv}} = \dfrac{1}{hA}$.
  \item \textbf{Five Core Dimensionless Groups ($\alpha = k/(\rho c_p)$, $\nu = \mu/\rho$):}
  \begin{itemize}
    \item \textbf{Nusselt ($\mathrm{Nu}$):} $\mathrm{Nu}_L = \dfrac{h L}{k_{\mathrm{fluid}}}$ (convection / pure fluid conduction).
    \item \textbf{Prandtl ($\mathrm{Pr}$):} $\mathrm{Pr} = \dfrac{\nu}{\alpha} = \dfrac{\mu c_p}{k}$ ($\delta_v/\delta_T$; air $\approx 0.71$, water $\SI{20}{\celsius}\approx 7.01$).
    \item \textbf{Reynolds ($\mathrm{Re}$):} $\mathrm{Re}_L = \dfrac{\rho u_\infty L}{\mu} = \dfrac{u_\infty L}{\nu}$ (forced inertia / viscous).
    \item \textbf{Grashof ($\mathrm{Gr}$) \& Rayleigh ($\mathrm{Ra}$):} $\mathrm{Gr}_L = \dfrac{g_n\beta(T_s - T_\infty)L^3}{\nu^2}$, $\boxed{\mathrm{Ra}_L = \mathrm{Gr}_L\,\mathrm{Pr} = \frac{g_n\beta(T_s - T_\infty)L^3}{\nu\alpha}}$ (buoyancy-driven natural convection!).
  \end{itemize}
  \item \textbf{Enclosed Dead-Air Gap Onset ($\mathrm{Ra}_\delta < 1708$):} In a horizontal cavity of thickness $\delta$, natural convection rolls stay suppressed for $\mathrm{Ra}_\delta < 1708$ ($\delta_{\mathrm{opt}} \approx \mathbf{10\text{--}\SI{12}{\milli\meter}}$; wider gaps convect, narrower gaps conduct as $k_{\mathrm{air}}/\delta$).
\end{itemize}
\end{formulacard}

\begin{formulacard}[Blackbody Radiation, Wien, Stefan--Boltzmann \& Shields \src{S12,S25}]
\scriptsize
\begin{itemize}
  \item \textbf{Planck Spectral Exitance \& Lambertian Radiance (\texttt{Cornell 2027 Q36}):}
  \[
    M_{e,\lambda} = \frac{2\pi h c_0^2}{\lambda^5(e^{hc_0/(\lambda k_B T)} - 1)},\; M_e = \sigma T^4 = \pi L_{\mathrm{rad}} \Rightarrow \boxed{L_{\mathrm{rad}} = \frac{\sigma T^4}{\pi}}
  \]
  \item \textbf{Wien's Displacement Law:} $\boxed{\lambda_{\max} T = b = \SI{2897.77}{\micro\meter\kelvin}}$ (Sun $\SI{5778}{\kelvin} \to \SI{0.501}{\micro\meter}$; $\SI{2400}{\kelvin}$ heat lamp $\to \mathbf{\SI{1.207}{\micro\meter}}$ near-IR; $\SI{300}{\kelvin}$ water $\to \mathbf{\SI{9.66}{\micro\meter}}$ mid-IR).
  \item \textbf{Stefan--Boltzmann Net Radiation \& Linearized $h_{\mathrm{rad}}$ ($T$ in $\mathbf{\si{\kelvin}}$!):}
  \[
    \boxed{\dot{Q}_{\mathrm{rad,net}} = \varepsilon\sigma A\left(T_s^4 - T_{\mathrm{surr}}^4\right) \approx h_{\mathrm{rad}}A(T_s - T_{\mathrm{surr}})}, \qquad h_{\mathrm{rad}} \approx 4\varepsilon\sigma T_{\mathrm{avg}}^3
  \]
  \item \textbf{Surface Balance \& Kirchhoff's Law:} $\alpha + \rho + \tau = 1$ (opaque: $\alpha + \rho = 1$); at equal $\lambda, T$: $\alpha_\lambda = \varepsilon_\lambda$. \textbf{$N$ Parallel Reflective Shields (emissivity $\varepsilon$):}
  \[
    \boxed{\dot{Q}_{N\text{ shields}} = \frac{1}{N + 1}\,\dot{Q}_{0} = \frac{\sigma A(T_1^4 - T_2^4)}{(N + 1)\left(\frac{2}{\varepsilon} - 1\right)}}
  \]
  \item \textbf{Planetary Radiative Equilibrium (\texttt{MIT 2018 Q23}):} $T_p = T_{\mathrm{star}}\left(\frac{1 - a_{\mathrm{albedo}}}{4}\right)^{1/4}\sqrt{\frac{R_{\mathrm{star}}}{D}}$.
\end{itemize}
\end{formulacard}

\begin{twocol}[0.50]
\begin{formulacard}[Lumped Capacitance ($\mathrm{Bi}<0.1$) \& 2027 Lamp Heating Curve \src{S12}]
\scriptsize
\begin{itemize}
  \item \textbf{Biot Number Validity Criterion ($L_c = V/A_s$):}
  \[
    \boxed{\mathrm{Bi} = \frac{h L_c}{k_{\mathrm{solid}}} = \frac{R_{\mathrm{cond,internal}}}{R_{\mathrm{conv,external}}} < 0.1} \quad (\text{Fourier number: } \mathrm{Fo} = \frac{\alpha t}{L_c^2}, \ \frac{t}{\tau} = \mathrm{Bi}\cdot\mathrm{Fo})
  \]
  ($L_c = L$ for 1-sided slab; $R/2$ for cylinder; $R/3$ for sphere).
  \item \textbf{Newton Exponential Cooling / Oven Heating ($\dot{Q}_{\mathrm{in}} = 0$):}
  \[
    \boxed{\theta(t) = \frac{T(t) - T_\infty}{T_i - T_\infty} = e^{-t/\tau}}, \quad \boxed{\tau = R_{\mathrm{th}}C_{\mathrm{th}} = \frac{mc}{hA_s}}, \quad t = \tau\ln\frac{T_i - T_\infty}{T(t) - T_\infty}
  \]
  \item \textbf{2027 Collector Heating Under Constant Absorbed Power $\dot{Q}_{\mathrm{abs}}$:}
  With total heat-loss conductance $G_{\mathrm{loss}} = 1/R_{\mathrm{th}} = U A_s$ to ambient $T_\infty$, the water approaches steady-state stagnation temperature $T_{\mathrm{ss}} = T_\infty + \dot{Q}_{\mathrm{abs}}R_{\mathrm{th}}$:
  \[
    \boxed{T(t) = T_{\mathrm{ss}} - (T_{\mathrm{ss}} - T_i)e^{-t/\tau} = T_i + \dot{Q}_{\mathrm{abs}}R_{\mathrm{th}}\left(1 - e^{-t/\tau}\right)}, \qquad \tau = R_{\mathrm{th}}C_{\mathrm{eff}}
  \]
  \emph{Short-time limit ($t \ll \tau$):} $1 - e^{-t/\tau} \approx \frac{t}{\tau} - \frac{1}{2}\left(\frac{t}{\tau}\right)^2 \implies \Delta T(t) \approx \frac{\dot{Q}_{\mathrm{abs}}}{C_{\mathrm{eff}}}\,t\left(1 - \frac{t}{2\tau}\right)$.
  \item \textbf{Semi-Infinite Solid Diffusion Depth:} $\delta_{\mathrm{pen}} \approx \sqrt{4\alpha t}$; surface step response depends on \textbf{thermal effusivity} $e_{\mathrm{th}} = \sqrt{k\rho c_p}$ (why metal feels colder than wood at the same room temperature!).
\end{itemize}
\end{formulacard}

\begin{formulacard}[Thermometer Physics, $5\tau_p$ Lag Rule \& Immersion Bias \src{S21,S22}]
\scriptsize
\begin{itemize}
  \item \textbf{First-Order Probe Step Response ($\tau_p = m_p c_p / (h_p A_p)$):}
  \[
    \frac{T_{\mathrm{true}} - T_{\mathrm{p}}(t)}{T_{\mathrm{true}} - T_{\mathrm{p},0}} = e^{-t/\tau_p} \Rightarrow \begin{cases} 1\tau_p: & 63.2\%\text{ settled } (36.8\%\text{ err})\\ 3\tau_p: & 95.0\%\text{ settled } (5.0\%\text{ err})\\ \mathbf{5\tau_p}: & \mathbf{99.33\%\text{ settled } (<0.67\%\text{ err})} \end{cases}
  \]
  \item \textbf{5 Thermometer Families \& Critical Traps:}
  \begin{enumerate}
    \item \textbf{Pt100 RTD:} $R(T) = R_0(1 + a T_C)$ ($R_0 = \SI{100.00}{\ohm}$ at $\SI{0}{\celsius}$, $a = \SI{3.850e-3}{\per\celsius}$); highest linearity/accuracy, requires 4-wire Kelvin lead compensation.
    \item \textbf{NTC Thermistor:} $R(T) = R_0\exp\left[\beta\left(\frac{1}{T} - \frac{1}{T_0}\right)\right]$ ($\beta \approx \SI{3950}{\kelvin}$); highest sensitivity, nonlinear, self-heating $I^2R / G_{\mathrm{th}}$.
    \item \textbf{Thermocouple (Types K/T, NIST ITS-90 \src{S22}):} Seebeck EMF $\Delta V = S_{AB}(T_{\mathrm{meas}} - T_{\mathrm{ref}})$ ($\sim\SI{40}{\micro\volt\per\kelvin}$); measures $\Delta T$ only, requires cold-junction reference compensation; fastest $\tau_p$ ($\SIrange{0.1}{1.5}{\second}$).
    \item \textbf{IR Pyrometer Trap:} Measures \textbf{only the top $\sim\SI{10}{\micro\meter}$ surface skin} ($\varepsilon_{\mathrm{water}} \approx 0.96$ in $8\text{--}\SI{14}{\micro\meter}$) and cannot see through glass or plastic!
    \item \textbf{2027 Immersion Depth Bias (FAQ 10/06/2026 on probe depth):} Because near-IR lamp radiation heats the top $\SI{10}{\milli\meter}$ of water, probing at $z = \SI{8}{\milli\meter}$ reads $1.25\text{--}1.45\times$ higher $\mathrm{TG}$ than probing at the bottom ($z = \SI{35}{\milli\meter}$).
  \end{enumerate}
\end{itemize}
\end{formulacard}
