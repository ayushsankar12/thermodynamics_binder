\begin{rulebox}[CRITICAL 2027 DEVICE RULES \& HARD SAFETY BAN (Rule 3.b.i--iii \& the 11 official FAQs)]
\begin{itemize}
  \item \textbf{2027 Functional Objective (Rules~\texttt{1, 3.a, 4.a--g}):} Collect downward radiant energy from an ES-provided \textbf{$\SIrange{100}{125}{\watt}$ untinted non-LED/halogen/ceramic heat lamp} at \textbf{$h_{\mathrm{bulb}} = \SIrange{40}{50}{\centi\meter}$ (Reg) / $\SIrange{40}{55}{\centi\meter}$ (State) / $\SIrange{40}{65}{\centi\meter}$ (Natl)} to heat \textbf{$\SI{100}{\milli\liter}$ of room-temperature water} in an ES-provided \textbf{$\SI{9}{oz}$ clear plastic party cup} ($\approx\SI{10.0}{\centi\meter}$ tall, $\SI{7.5}{\centi\meter}$ mouth) inside a \textbf{$\SI{30.0}{\centi\meter}\times\SI{30.0}{\centi\meter}\times\SI{30.0}{\centi\meter}$ bounding cube} for \textbf{$\SI{10.0}{\minute}$ (Reg) or $\SIrange{10}{15}{\minute}$ (State/Natl)}.
  \item \textbf{\textcolor{rust}{ZERO-TOLERANCE POLYSTYRENE / STYROFOAM / \texttt{\#6~PS} BAN (Rule~\texttt{3.b.i,iii} \& FAQ 10/22/2026 on polystyrene, \texttt{10/22/2026}):}} \textbf{Polystyrene in all forms} (Styrofoam, EPS bead foam, XPS blue/pink/green board, craft foam-core posterboard with PS core, and rigid \texttt{\#6~PS} plastics), alongside \textbf{fiberglass, mineral wool, and asbestos}, is classified as a \textbf{Rule~\texttt{3.b.i} hazardous material under a heat lamp}. Any device containing these materials \textbf{will NOT be allowed to be set up, modified, or tested ($\mathbf{\mathrm{TS}=0, \mathrm{PS}=0}$)}!
  \item \textbf{Zero Water Contact, Removable Glazing \& Zero Swirling (Rule~\texttt{3.e, 5.II.d,g} \& the official FAQs (09/30--10/22/2026)):} No part of the device may contact the water (\texttt{3.e}); nothing may be added to the water (\texttt{3.e}); no adhesives, tape, rubber bands, or clips may be attached to the cup (FAQ 10/21/2026 on items on the cup), though a gravity-resting clear BoPET glazing window that is completely removed before $T_{\mathrm{final}}$ measurement is legal (FAQs 10/21 and 10/22/2026 on the cup and lid); \textbf{neither students nor the ES may stir, swirl, or shake the water} (FAQs 09/30, 10/01 and 10/06/2026 on stirring and swirling), and the ES probes the \textbf{top water layer near the center of the cup} (FAQ 10/06/2026 on probe depth).
\end{itemize}
\end{rulebox}

\begin{devicebox}[Five-Phase Competition Block Rapid Execution Summary (Partner B Device Lead Protocol)]
\begin{enumerate}
  \item \textbf{Morning Impound (\texttt{Rule 2.d}, FAQ 10/28/2026 on impound):} Equilibrate open collector to room temp ($\ge\SI{20}{\minute}$); impound \textbf{only} physical device + spare BoPET glazing rings. Keep this handbook, calculators (\texttt{2.b}), and plumb card with team!
  \item \textbf{3-Min Device Setup (\texttt{Rule 5.II.a--b}, Lamp OFF):} Read posted $h_{\mathrm{bulb}}, t_{\mathrm{heat}}, T_{\mathrm{room}}, P_{\mathrm{bulb}}$. Lock pedestal into \textbf{Slot~A} ($\SIrange{40}{48}{\centi\meter}$), \textbf{Slot~B} ($\SIrange{49}{57}{\centi\meter}$), or \textbf{Slot~C} ($\SIrange{58}{65}{\centi\meter}$); align crosshairs under cold bulb; look up $\mathrm{TG}_{\mathrm{table}}(h, P, t)$ and $k_{\mathrm{room}}(t)$ from \tref{tab:16:doe-36} / \tref{tab:16:natl-1cm}.
  \item \textbf{1-Min Water Setup (\texttt{Rule 5.II.c--d}, Lamp OFF):} Receive $\SI{9}{oz}$ cup ($\SI{100}{\milli\liter}$) and $T_{\mathrm{start}}$ from ES. Seat cup on 3-fin cork cradle ($\SI{3}{\second}$), lower upper reflector hood + gravity BoPET window onto 4 guide dowels ($\SI{4}{\second}$), verify centering, and step back $\ge\SI{1.0}{\meter}$!
  \item \textbf{Heating Window (\texttt{Rule 5.II.e--f}, $\SIrange{10}{15}{\minute}$, Lamp ON):} \textbf{Hands off!} Compute $\TpredFormula$ to $\SI{0.1}{\celsius}$, verify on the Class~III calculator (\texttt{PRGM:THERMO27}), and hand it in: \textbf{\HandInRule} Then join Partner~A on the written exam.
  \item \textbf{1-Min Final Measurement (\texttt{Rule 5.II.g--h}, Lamp OFF):} Wait until ES is at station ready to probe; lift upper hood + BoPET cap straight up off dowels ($\SI{2}{\second}$); lift bare cup vertically by top rim with \textbf{zero tilt or swirl} ($\SI{4}{\second}$) and place on Testing Surface. Verify recorded scoresheet data!
\end{enumerate}
\end{devicebox}
