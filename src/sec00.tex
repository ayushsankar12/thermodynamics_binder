\section{2027 Scoring Engine, Penalty Stacking \& Tiebreaker Mathematics}\label{sec:00.2}
\idxtopic{scoring formulas (2027)}
\idxformula{Final Score FS = ES + TS + PS}
\idxformula{Temperature Score TS and Temperature Gain TG}
\idxformula{Prediction Score PS and Prediction Estimate PE}
\idxformula{Penalty multiplier M_pen = 0.7^Vc * 0.9^Vp}

\begin{formulacard}[Official 2027 Scoring Equations, Penalty Multipliers \& Tiebreakers (Rule 6.a--j \& Scoresheet v2.0.0 {\src{S01,S04}})]
\setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}
All temperature inputs and calculations are strictly in \textbf{degrees Celsius ($\si{\celsius}$)} (\texttt{Rule 6.b}). High score wins ($\max \mathrm{FS} = 100.0\text{ pts}$):
\begin{equation}\label{eq:00:fs}
  \mathrm{FS} = \mathrm{ES} + \mathrm{TS} + \mathrm{PS}, \quad
  \mathrm{ES} = 50 \times \frac{\text{Part I Score}}{\text{Highest Part I}}, \quad
  \mathrm{TS} = 30 \times M_{\mathrm{pen}} \times \frac{\mathrm{TG}}{\text{Highest }\mathrm{TG}}, \quad
  \mathrm{PS} = 20 \times M_{\mathrm{pen}} \times \frac{\mathrm{PE}}{\text{Highest }\mathrm{PE}}
\end{equation}
\begin{equation}\label{eq:00:tg-pe}
  \mathrm{TG} = \max\!\left(0,\; T_f - T_i\right)\ [\si{\celsius}], \qquad
  \mathrm{PE} = \max\!\left(0,\; 1 - \frac{\left|T_f - T_{\mathrm{pred}}\right|}{T_f}\right)\ \left(T \text{ in }\si{\celsius}\right), \qquad
  M_{\mathrm{pen}} = (0.7)^{V_c} \times (0.9)^{V_p}
\end{equation}
\begin{itemize}
  \item \textbf{Construction Violations ($V_c$, $\times 0.7$ each to $\mathrm{TS},\mathrm{PS}$, Rule \texttt{6.h} \& Box 3):} Unfixed violations at end of 3-min Setup across \texttt{2.d} (non-impound/temp \src{S02}), \texttt{3.b.ii} (probes/thermoses/coolers/vacuum), \texttt{3.c} (9 oz cup fit), \texttt{3.d} ($\SI{30.0}{\centi\meter}$ cube), \texttt{3.e} (water contact/additives), \texttt{3.g} (room temp), \texttt{3.h} (Building Policy).
  \item \textbf{Competition Violations ($V_p$, $\times 0.9$ each to $\mathrm{TS},\mathrm{PS}$, Rule \texttt{6.g} \& Box 5):} Violations of \texttt{5.II.b} (post-setup modification beyond cup insert/remove) or \texttt{5.II.d} (altering cup/water, tape/adhesives/clips on cup \src{S02}, or swirling/stirring water \src{S02}).
  \item \textbf{Official Tiebreakers (\texttt{Rule 6.j} \src{S04}):} \textbf{1st (\texttt{TB1}): Best $\mathrm{TS}$}; \textbf{2nd (\texttt{TB2}): Best $\mathrm{ES}$}; \textbf{3rd (\texttt{TB3}): Preselected written test questions}.
\end{itemize}
\end{formulacard}

\begin{workedexample}[Clean Run vs.\ Stacked Penalty Audit on a 55-Team Invitational Field]{2027 Rule 6.c--h}{ex:00.1}
\setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}
At an invitational where field highs are $\max(\text{Part I}) = 85.00\text{ pts}$ (out of $238.0$, Cornell 2027 \texttt{E1} \src{S25}), $\max(\mathrm{TG}) = \SI{16.0}{\celsius}$, and $\max(\mathrm{PE}) = 0.9950$, \textbf{Team Alpha} scores $78.20\text{ pts}$ on Part I, starts at $T_i = \SI{21.4}{\celsius}$, predicts $T_{\mathrm{pred}} = \SI{36.3}{\celsius}$, and reaches $T_f = \SI{36.6}{\celsius}$ ($V_c = V_p = 0$). \textbf{Team Beta} achieves identical raw numbers but failed to impound a spare lid ($V_c = 1$, \texttt{2.d} \src{S02}) and taped cling wrap to the cup ($V_p = 1$, \texttt{5.II.d} \src{S02}). Compute both Final Scores.
\par\smallskip\noindent
\textbf{Solution.} For \textbf{Team Alpha} ($M_{\mathrm{pen}} = 1.0000$): $\mathrm{ES}_\alpha = 50(78.20/85.00) = 46.0000\text{ pts}$, $\mathrm{TG}_\alpha = 36.6 - 21.4 = \SI{15.2}{\celsius} \implies \mathrm{TS}_\alpha = 30(15.2/16.0) = 28.5000\text{ pts}$, and $\mathrm{PE}_\alpha = 1 - 0.3/36.6 = 0.991803 \implies \mathrm{PS}_\alpha = 20(0.991803/0.9950) = 19.9357\text{ pts}$, giving $\mathrm{FS}_\alpha = 46.0000 + 28.5000 + 19.9357 = \mathbf{94.4357 \approx 94.44\text{\textbf{ pts}}}$.
For \textbf{Team Beta}, $M_{\mathrm{pen},\beta} = (0.7)^1(0.9)^1 = 0.6300$ multiplies $\mathrm{TS}$ and $\mathrm{PS}$ ($\mathrm{ES}_\beta = 46.0000$ unchanged): $\mathrm{TS}_\beta = 28.5000(0.63) = 17.9550\text{ pts}$ and $\mathrm{PS}_\beta = 19.935744(0.63) = 12.5595\text{ pts}$, yielding $\mathrm{FS}_\beta = \mathbf{76.5145 \approx 76.51\text{\textbf{ pts}}}$ (a loss of $\mathbf{17.92\text{\textbf{ overall points}}}$!).
\verifiedby{Independent Python check \texttt{SEC00\_EX01\_CLEAN\_SCORE} \& \texttt{SEC00\_EX01\_PENALTY\_STACK} against Scoresheet v2.0.0 formulas.}
\end{workedexample}

\begin{workedexample}[Tiebreaker Resolution (\texttt{TB1 = Best TS}) \& The Celsius Denominator in $\mathrm{PE}$]{2027 Rule 6.b, 6.f, 6.j}{ex:00.2}
(a) \textbf{Team Gamma} ($\mathrm{ES}=44.00, \mathrm{TS}=29.00, \mathrm{PS}=18.50$) and \textbf{Team Delta} ($\mathrm{ES}=45.50, \mathrm{TS}=27.50, \mathrm{PS}=18.50$) tie at $\mathrm{FS} = 91.50\text{ pts}$. Who wins? (b) At $T_f = \SI{32.0}{\celsius}$ ($\max\mathrm{PE}=1.000, M_{\mathrm{pen}}=1$), how many $\mathrm{PS}$ points does a $\SI{0.40}{\celsius}$ prediction error cost, and by what factor does a Kelvin denominator misestimate it?
\par\smallskip\noindent
\textbf{Solution.} (a) Under Rule \texttt{6.j}, \textbf{1st Tiebreaker (\texttt{TB1}) is Best $\mathrm{TS}$}, \emph{not} Best $\mathrm{ES}$. Because $\mathrm{TS}_\gamma = 29.00 > \mathrm{TS}_\delta = 27.50$, \textbf{Team Gamma wins the tie} despite a lower written score.
(b) In Celsius (\texttt{Rule 6.b}), $\Delta\mathrm{PE} = |\Delta T_{\mathrm{err}}|/T_f = 0.40/32.0 = 0.01250$, costing $\Delta\mathrm{PS} = 20(0.01250) = \mathbf{0.250\text{\textbf{ pts}}}$. Using $T_f = \SI{305.15}{\kelvin}$ gives $\Delta\mathrm{PE}_K = 0.001311$, underestimating point sensitivity by $305.15/32.0 = \mathbf{9.536\times}$.
\verifiedby{Independent Python check \texttt{SEC00\_EX02\_TIEBREAKER\_AND\_PE\_TRAP} (\texttt{verify/checks/check\_sec00.py}).}
\end{workedexample}
